{% comment %} Language switcher (immersive#612), rendered at the bottom of the sidebar on every page. The six editions mirror the English page structure, so a page maps to its counterpart by swapping the locale prefix. Clicking a language stores it as the explicit preference, which head_custom.html's browser-language detection then always respects. {% endcomment %} {% assign locale_codes = "ca|es|fr|it|pt" | split: "|" %} {% assign locale_names = "Català|Español|Français|Italiano|Português" | split: "|" %} {% assign url_parts = page.url | split: "/" %} {% assign current_locale = "en" %} {% assign page_suffix = page.url %} {% for code in locale_codes %} {% if url_parts[1] == code %} {% assign current_locale = code %} {% assign prefix_size = code.size | plus: 1 %} {% assign page_suffix = page.url | slice: prefix_size, page.url.size %} {% endif %} {% endfor %}
